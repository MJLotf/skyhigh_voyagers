-- 1. Create the pilots table
CREATE TABLE pilots (
    safa_number INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    glider_class TEXT CHECK (glider_class IN ('A', 'B', 'C', 'D', 'CCC'))
);

-- 2. Create the tasks table
CREATE TABLE tasks (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    xctrack_json_data JSONB,
    max_score NUMERIC CHECK (max_score BETWEEN 0 AND 1000),
    designer TEXT,
    description TEXT
);

-- 3. Create the flights table
CREATE TABLE flights (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    pilot_id INT REFERENCES pilots(safa_number) ON DELETE CASCADE,
    task_id INT REFERENCES tasks(id) ON DELETE CASCADE,
    speed_score NUMERIC CHECK (speed_score BETWEEN 0 AND 1000),
    distance_score NUMERIC CHECK (distance_score BETWEEN 0 AND 1000),
    total_score NUMERIC CHECK (total_score BETWEEN 0 AND 1000),
    participants INT CHECK (participants >= 1)
);
